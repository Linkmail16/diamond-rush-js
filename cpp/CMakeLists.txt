cmake_minimum_required(VERSION 3.16)
project(diamondrush CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

file(GLOB GAME_SRC src/game/*.cpp)
add_executable(diamondrush ${GAME_SRC} src/runtime/runtime.cpp src/runtime/platform.cpp src/res/res_data.cpp)

if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
	target_compile_options(diamondrush PRIVATE -fwrapv -fno-strict-aliasing -w)
	set_source_files_properties(src/res/res_data.cpp PROPERTIES COMPILE_OPTIONS "-O0")
elseif(MSVC)
	target_compile_options(diamondrush PRIVATE /bigobj /W0 /utf-8 /d2SSAOptimizer-)
endif()

if(SDL2_MINGW_DIR)
	target_include_directories(diamondrush PRIVATE ${SDL2_MINGW_DIR}/include/SDL2)
	target_link_directories(diamondrush PRIVATE ${SDL2_MINGW_DIR}/lib)
	target_compile_definitions(diamondrush PRIVATE SDL_MAIN_HANDLED)
	target_link_libraries(diamondrush PRIVATE -static mingw32 SDL2main SDL2 -mwindows
		m dinput8 dxguid dxerr8 user32 gdi32 winmm imm32 ole32 oleaut32 shell32 setupapi version uuid)
else()
	find_package(SDL2 REQUIRED)
	if(TARGET SDL2::SDL2)
		target_link_libraries(diamondrush PRIVATE SDL2::SDL2)
		if(TARGET SDL2::SDL2main)
			target_link_libraries(diamondrush PRIVATE SDL2::SDL2main)
		endif()
	else()
		target_include_directories(diamondrush PRIVATE ${SDL2_INCLUDE_DIRS})
		target_link_libraries(diamondrush PRIVATE ${SDL2_LIBRARIES})
	endif()
endif()
